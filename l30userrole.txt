=> Update Role

mongosh -u rootadmin -p --authenticationDatabase admin
mongodbcloud123

use proone

db.createUser({
    user: "reportuser",
    pwd: "123456789",
    roles: [
        {role: "read", db: "proone"}
    ]
})

db.updateUser("reportuser",{
    pwd: "987654321",
    roles: [
        {role: "readWrite", db: "proone"}
    ]
})


db.grantRolesToUser("reportuser",[
    {role: "readWrite", db: "proone"}
]);

use proone
db.getUsers();

exit 

=Test
mongosh -u reportuser -p --authenticationDatabase proone
987654321

exit
