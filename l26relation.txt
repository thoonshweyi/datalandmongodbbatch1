//= exe1 Joins posts with users

db.posts.aggregate([
    {
        $lookup:{
            from: "users",
            localField: "userId",
            foreignField: "_id",
            as: "author"
        }
    },
    {
        $project: {
            title:1,
            content:1,
            "author.name":1,
            "author.email":1
        }
    }
])

//= exe2 Joins posts with users

db.posts.aggregate([
    {
        $lookup:{
            from: "users",
            localField: "userId",
            foreignField: "_id",
            as: "author" // output array field
        }
    },
    {
        $unwind: "$author" // convert array to object
    },
    {
        $project: {
            title:1,
            content:1,
            "author.name":1,
            "author.email":1
        }
    }
])
