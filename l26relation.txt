=> Lookup with pipline, $match
// = exe 3 (i)
db.posts.aggregate([
    {
        $lookup:{
            from: "users",
            let: {authId: "$userId"},
            pipeline: [
                {
                    $match: {
                        $expr: {$eq: ["$_id","$$authId"]}
                    }
                }
            ],
            as: "author"
        }
    },
])


// = exe 3 (ii)
db.posts.aggregate([
    {
        $lookup:{
            from: "users",
            let: {authId: "$userId"},
            pipeline: [
                {
                    $match: {
                        $expr: {$eq: ["$_id","$$authId"]}
                    }
                }
            ],
            as: "author"
        }
    },
    {$unwind: "$author"}
])
